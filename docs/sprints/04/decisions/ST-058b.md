## Mechanical checklist check (C2, C3, C4, C5 trigger words, C7, C8; AN-02 and AN-04 with ≥ 2 drills at every level)

Run from the repository root: `python3 -I st058b_check.py content/drills <the 7 ids, comma separated>` with this script saved as `st058b_check.py` (not committed: the ticket may change only `content/drills/` and this file). C1 is `racket-drill-lint`; C5 (beyond the trigger words), C6 and C7 are read by the coach (rows below).

```python
"""ST-058b mechanical checklist (drill-review-checklist C2, C3, C4, C5 trigger words, C7, C8; AN-02/AN-04 cells)."""
import json, re, sys
from pathlib import Path
root = Path(sys.argv[1])
want = sys.argv[2].split(",")
R1 = {"AN-01", "AN-02", "AN-04", "AN-05", "AN-07"}
LV = ["beginner", "intermediate", "advanced"]
probs = []; cells = {}
for d in want:
    p = root / d / "v1.json"
    if not p.exists():
        probs.append(f"{d}: missing"); continue
    j = json.loads(p.read_text())
    if j["id"] != d: probs.append(f"{d}: id/dir mismatch")
    if "warm-up" in j["skills"]: probs.append(f"{d}: ST-058b adds no warm-up")
    t = set(j["target_metrics"])
    if not t <= R1: probs.append(f"{d}: C3 non-R1 target")
    if not t & {"AN-02", "AN-04"}: probs.append(f"{d}: not AN-02/AN-04")
    if ("AN-07" in t) != any(s.startswith("share:") for s in j["skills"]): probs.append(f"{d}: C3 share rule")
    if not re.search(r"\d", j["success_criterion"]) or re.search(r"improve|feel|until comfortable", j["success_criterion"], re.I):
        probs.append(f"{d}: C4")
    if "(judgment)" not in j.get("coach_rationale", "") and not j.get("source"): probs.append(f"{d}: C2")
    text = " ".join([j["summary"], *j["setup"]]).lower()
    if re.search(r"lob|speed-up|live|sprint|dive", text) and not j["safety_notes"].strip(): probs.append(f"{d}: C5 safety_notes empty")
    if j["players"] > 2 or "ball_machine" in j["needs"]: probs.append(f"{d}: C7 not solo/pairs or needs ball machine")
    if not (1 <= j["duration_min"] <= j["duration_max"] <= 45): probs.append(f"{d}: C8")
    for lv in LV[LV.index(j["level_min"]):LV.index(j["level_max"]) + 1]:
        for m in t & {"AN-02", "AN-04"}: cells.setdefault((m, lv), []).append(d)
for m in ("AN-02", "AN-04"):
    for lv in LV:
        n = len(cells.get((m, lv), []))
        print(f"{m} x {lv}: drills {n}")
        if n < 2: probs.append(f"{m} x {lv}: {n} < 2")
if not any(json.loads((root / d / "v1.json").read_text())["players"] == 1 for d in want if (root / d / "v1.json").exists()):
    probs.append("no solo drill")
for x in probs: print("FAIL", x)
print(f"st058b-checklist: {len(probs)} problem(s)")
sys.exit(1 if probs else 0)
```
