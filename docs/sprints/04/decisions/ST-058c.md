## Mechanical checklist and coverage matrix (C2, C3, C4, C5 trigger words, C7, C8 for the new drills; §3 for the whole library)

Run from the repository root: `python3 -I st058c_check.py content/drills <the 9 ids, comma separated>` with this script saved as `st058c_check.py` (not committed: the ticket may change only `content/drills/` and this file). The matrix part implements drill-review-checklist §3 over the whole library (latest version, `coach-reviewed`/`verified`, no `deprecated_by`, not a warm-up; AN-07 by share skill). C1 is `racket-drill-lint`; C5 (beyond the trigger words), C6 and C7 are read by the coach (rows below).

```python
"""ST-058c mechanical checklist (C2, C3, C4, C5 trigger words, C7, C8 for the new drills) and the
full COACH-4 coverage matrix (drill-review-checklist §3: 6 targets x 3 levels, >= 20 reviewed, >= 3 warm-ups)."""
import json, re, sys
from pathlib import Path
root = Path(sys.argv[1])
want = sys.argv[2].split(",")
R1 = {"AN-01", "AN-02", "AN-04", "AN-05", "AN-07"}
LV = ["beginner", "intermediate", "advanced"]
TG = ["AN-01", "AN-02", "AN-04", "AN-05", "AN-07:fault", "AN-07:forced-error"]
probs = []
for d in want:
    p = root / d / "v1.json"
    if not p.exists():
        probs.append(f"{d}: missing"); continue
    j = json.loads(p.read_text())
    if j["id"] != d: probs.append(f"{d}: id/dir mismatch")
    if "warm-up" in j["skills"]: probs.append(f"{d}: ST-058c adds no warm-up")
    t = set(j["target_metrics"])
    if not t <= R1: probs.append(f"{d}: C3 non-R1 target")
    if "AN-07" not in t: probs.append(f"{d}: does not target AN-07")
    if ("AN-07" in t) != any(s.startswith("share:") for s in j["skills"]): probs.append(f"{d}: C3 share rule")
    if not re.search(r"\d", j["success_criterion"]) or re.search(r"improve|feel|until comfortable", j["success_criterion"], re.I):
        probs.append(f"{d}: C4")
    if "(judgment)" not in j.get("coach_rationale", "") and not j.get("source"): probs.append(f"{d}: C2")
    text = " ".join([j["summary"], *j["setup"]]).lower()
    if re.search(r"lob|speed-up|drive|live|sprint|dive", text) and not j["safety_notes"].strip(): probs.append(f"{d}: C5 safety_notes empty")
    if j["players"] > 2 or "ball_machine" in j["needs"]: probs.append(f"{d}: C7 not solo/pairs or needs ball machine")
    if not (1 <= j["duration_min"] <= j["duration_max"] <= 45): probs.append(f"{d}: C8")
# coverage over the whole library (latest version, reviewed, not deprecated, not a warm-up)
latest = {}
for p in sorted(root.glob("*/v*.json")):
    j = json.loads(p.read_text())
    if j["id"] not in latest or j["version"] > latest[j["id"]]["version"]: latest[j["id"]] = j
reviewed = [j for j in latest.values() if j["review_status"] in ("coach-reviewed", "verified") and not j.get("deprecated_by")]
warm = [j for j in reviewed if "warm-up" in j["skills"]]
cells = {}
for j in reviewed:
    if "warm-up" in j["skills"]: continue
    hits = set(j["target_metrics"]) - {"AN-07"}
    if "AN-07" in j["target_metrics"]:
        sh = {s[6:] for s in j["skills"] if s.startswith("share:")}
        if not sh: probs.append(f"{j['id']}@{j['version']}: AN-07 without share skill")
        hits |= {f"AN-07:{s}" for s in sh}
    for lv in LV[LV.index(j["level_min"]):LV.index(j["level_max"]) + 1]:
        for m in hits: cells.setdefault((m, lv), []).append(j)
print(f"{'target':<20}{'level':<14}{'drills':>7}{'players<=2':>12}{'no ball_machine':>17}")
for m in TG:
    for lv in LV:
        c = cells.get((m, lv), [])
        n, p2, nb = len(c), sum(x["players"] <= 2 for x in c), sum("ball_machine" not in x["needs"] for x in c)
        print(f"{m:<20}{lv:<14}{n:>7}{p2:>12}{nb:>17}")
        if n < 2: probs.append(f"{m} x {lv}: drills: {n}")
        if p2 < 1: probs.append(f"{m} x {lv}: players <= 2: 0")
        if nb < 1: probs.append(f"{m} x {lv}: no ball_machine: 0")
print(f"library: {len(reviewed)} reviewed drill(s), {len(warm)} warm-up(s)")
if len(reviewed) < 20: probs.append(f"library: {len(reviewed)} < 20 reviewed drills")
if len(warm) < 3: probs.append(f"warm-ups: {len(warm)} < 3")
for x in probs: print("FAIL", x)
print(f"st058c-checklist: {len(probs)} problem(s)")
sys.exit(1 if probs else 0)
```
