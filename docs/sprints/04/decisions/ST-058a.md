## Mechanical checklist check (C2, C3, C4, C8, warm-up rule §2, ≥ 3 warm-ups, AN-01 and AN-05 at every level)

Run from the repository root: `python3 -I st058a_check.py content/drills <the 7 ids, comma separated>` with this script saved as `st058a_check.py` (not committed: the ticket may change only `content/drills/` and this file). C1 is `racket-drill-lint`; C5, C6 and C7 are read by the coach (rows below).

```python
"""ST-058a mechanical checklist (drill-review-checklist C3, C4, C7, C8 and the warm-up rule §2)."""
import json, re, sys
from pathlib import Path
root = Path(sys.argv[1])
want = sys.argv[2].split(",")
R1 = {"AN-01", "AN-02", "AN-04", "AN-05", "AN-07"}
LV = ["beginner", "intermediate", "advanced"]
probs = []; warm = 0; serve = {}
for d in want:
    p = root / d / "v1.json"
    if not p.exists():
        probs.append(f"{d}: missing"); continue
    j = json.loads(p.read_text())
    w = "warm-up" in j["skills"]
    if w:
        warm += 1
        if not (j["duration_min"] >= 5 and j["duration_max"] <= 10 and j["level_min"] == "beginner"):
            probs.append(f"{d}: warm-up rule §2")
    else:
        if not set(j["target_metrics"]) <= R1: probs.append(f"{d}: C3 non-R1 target")
        if not set(j["target_metrics"]) & {"AN-01", "AN-05"}: probs.append(f"{d}: not AN-01/AN-05")
        for lv in LV[LV.index(j["level_min"]):LV.index(j["level_max"]) + 1]:
            for m in set(j["target_metrics"]) & {"AN-01", "AN-05"}: serve.setdefault(m, set()).add(lv)
    has07 = "AN-07" in j["target_metrics"]; share = any(s.startswith("share:") for s in j["skills"])
    if has07 != share: probs.append(f"{d}: C3 share rule")
    if not re.search(r"\d", j["success_criterion"]) or re.search(r"improve|feel|until comfortable", j["success_criterion"], re.I):
        probs.append(f"{d}: C4")
    if "(judgment)" not in j.get("coach_rationale", "") and not j.get("source"): probs.append(f"{d}: C2")
    if not j["id"] == d: probs.append(f"{d}: id/dir mismatch")
if warm < 3: probs.append(f"warm-ups: {warm} < 3")
for m in ("AN-01", "AN-05"):
    if serve.get(m, set()) != set(LV): probs.append(f"{m} levels: {sorted(serve.get(m, set()))}")
for x in probs: print("FAIL", x)
print(f"st058a-checklist: {len(probs)} problem(s), {warm} warm-up(s)")
sys.exit(1 if probs else 0)
```
