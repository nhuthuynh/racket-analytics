The Workflow-tool script that drove the 12-role team through Sprints 5 and 6, including the changes of 2026-11-14/15 (pr-check before every push, circuit breaker, ticket links, principal-engineer CI gate). Nothing in this repository runs it. To use it, copy the code block below to a `.js` file and pass that file to the Workflow tool (`scriptPath`). The sprint list it will run is embedded on line 15 of the script; read `runner-notes.md` first.

      if (fx && fx.policy_failure && ++policyFails >= 2) return { id: t.id, pr, merged: false, reason: 'blocked: PR-policy failure needs a decision outside the loop (twice)' }
      if (fx && fx.policy_failure) await agent(`${common(s)}
You are THE PR-POLICY DECISION STEP (engineering-manager for size, senior-qa-engineer for test immutability). PR #${pr} (ticket ${t.id}, branch ${branch}) fails 'PR policy'. Read the job log, decide on the merits as in working-agreement §5, record the decision in docs/sprints/${pad(s.n)}/decisions/${t.id}.md on the branch (worktree ${wt}), push, and apply the label only if justified. Comment the decision on the PR.`, { label: `${t.id}:policy:a${attempt}`, phase: phaseName })
      if (fx && (fx.pushed || fx.policy_failure || (fx.head_sha && fx.head_sha !== reviewed))) {
        const rr = await review(s, t, pr, branch, `a${attempt}`, ' This is a re-review after CI fixes: review what changed since your last review.')
        if (!rr.ok) {
          if (attempt === 5) return { id: t.id, pr, merged: false, reason: 'changes requested after CI fixes' }
          await agent(`${common(s)}
ROLE: ${t.owner}. Address the re-review findings on PR #${pr} in worktree ${wt}: ${JSON.stringify(rr.res.flatMap(x => x.findings.filter(f => /blocker|major/.test(f.severity))))}. Regression test first for behaviour bugs; push (no force); reply on each thread.`, { label: `${t.id}:refix:a${attempt}`, phase: phaseName })
          continue
        }
        reviewed = rr.res[0].reviewed_sha
      }
      continue
    }
    const m = await agent(`${common(s)}
ROLE: sre-devops-engineer acting as THE MERGE STEP (the only step allowed to merge). From a fresh origin/main checkout run 'python3 -I scripts/ci/merge_ready.py --repo ${OWNER}/${REPO} --pr ${pr} --sha ${gate.head_sha}'. Do not push or delete branches.`, { label: `${t.id}:merge:a${attempt}`, phase: phaseName, schema: { type: 'object', properties: { merged: { type: 'boolean' }, reason: { type: 'string' } }, required: ['merged', 'reason'] } })
    if (m && m.merged) return { id: t.id, pr, merged: true, reason: m.reason, ci: gate.checks }
    if (m && /Verdict|Reviewer|verdict/.test(m.reason)) {
      const missing = [...new Set([...m.reason.matchAll(/([a-z]+(?:-[a-z]+)+): no latest 'Verdict: APPROVE'/g)].map(x => x[1]).filter(r => ROLES.includes(r)))]
      const rr = await review(s, t, pr, branch, missing.length ? `m${attempt}-${missing.join('+')}` : `m${attempt}`, '', missing)
      if (rr.ok && !missing.length) reviewed = rr.res[0].reviewed_sha
      if (!rr.ok && missing.length) await agent(`${common(s)}
ROLE: ${t.owner}. Address the ${missing.join(', ')} review findings on PR #${pr} in worktree ${wt} (recreate from origin/${branch} if missing): ${JSON.stringify(rr.res.flatMap(x => x.findings.filter(f => /blocker|major/.test(f.severity))))}. Regression test first for behaviour bugs; push (no force); reply on each thread.`, { label: `${t.id}:mfix:a${attempt}`, phase: phaseName })
      continue
    }
    if (m && /conflict|not mergeable|405/i.test(m.reason)) continue
    if (attempt === 5) return { id: t.id, pr, merged: false, reason: m ? m.reason : 'merge agent failed' }
  }
  return { id: t.id, pr, merged: false, reason: 'not mergeable after 5 attempts' }
}

// PO 2026-11-15: a Blocked-by ticket outside this list counts only when verified merged into main
const extMerged = new Map()
async function mergedOnMain(id, phaseName) {
  if (!extMerged.has(id)) {
    const r = await agent(`ROLE: read-only checker (do not push, comment, label or merge). In ${OWNER}/${REPO}, is ticket ${id} merged into main? Search closed PRs whose title starts with '[${id}]' ('gh api "repos/${OWNER}/${REPO}/pulls?state=closed&per_page=100&page=N"', pages 1-5) and check merged_at. Return merged (true only if such a PR has merged_at set) and the PR number.`, { label: `dep:${id}`, phase: phaseName, effort: 'low', schema: { type: 'object', properties: { merged: { type: 'boolean' }, pr: { type: 'integer' } }, required: ['merged'] } })
    extMerged.set(id, !!(r && r.merged))
  }
  return extMerged.get(id)
}

// PO 2026-11-15: the principal-engineer monitors main CI; a red main is fixed before new tickets start (CI FIRST)
let ciFixN = 0
async function mainCiGreen(s, phaseName) {
  for (let k = 0; k < 2; k++) {
    const m = await agent(`ROLE: principal-engineer, CI/CD OWNER (PO 2026-11-15), read-only step: do not push, comment, label or merge. For ${OWNER}/${REPO}: list the latest COMPLETED run of workflow 'CI' on branch main for each event (push, schedule, workflow_dispatch) with 'gh api "repos/${OWNER}/${REPO}/actions/runs?branch=main&per_page=50"'. main is red if the latest completed push run or the latest completed scheduled run concluded failure. If red: read the failing job logs ('gh api repos/${OWNER}/${REPO}/actions/runs/<id>/attempts/1/jobs' then '/actions/jobs/<job>/logs'), root-cause it with the troubleshooting method of the CI RCA 2026-11-14 (examine, hypotheses with a disproving test, cure; 'flaky' is not a root cause), and check whether an open PR already fixes it. Return red, failing (job: step), root_cause, fix_plan, existing_fix_pr.`, { label: `ci-monitor:${ciFixN}:${k}`, phase: phaseName, schema: { type: 'object', properties: { red: { type: 'boolean' }, failing: { type: 'array', items: { type: 'string' } }, root_cause: { type: 'string' }, fix_plan: { type: 'string' }, existing_fix_pr: { type: 'integer' } }, required: ['red'] } })
    if (!m) return { ok: false, reason: 'CI monitor failed' }
    if (!m.red) return { ok: true }
    const id = `CI-FIX-${pad(s.n)}-M${++ciFixN}`
    log(`Sprint ${s.n}: main CI red (${(m.failing || []).join('; ')}); ${id} before new tickets`)
    const r = await deliverTicket(s, { id, title: `Fix red main CI: ${(m.failing || ['unknown'])[0]}`.slice(0, 80), owner: 'sre-devops-engineer', tier: 1, ci_fix: true, depends_on: [], blocked_by: [], relates_to: [],
      branch: `s${pad(s.n)}/ci-fix-main-m${ciFixN}`, existing_pr: m.existing_fix_pr || undefined,
      scope: `CI FIRST. Root cause from the principal-engineer (CI/CD owner): ${m.root_cause || 'see failing job logs'}. Failing: ${JSON.stringify(m.failing || [])}. Fix plan: ${m.fix_plan || 'root-cause from the log'}. Reproduce the failure locally first (red), then fix (green); never skip or weaken a test.` }, phaseName)
    if (!r || !r.merged) return { ok: false, reason: `main CI red and ${id} not merged: ${r ? r.reason : 'agent failure'}` }
  }
  return { ok: false, reason: 'main CI still red after two fixes' }
}

async function deliverAll(s, tickets, phaseName) {
  const done = new Map()
  tickets = tickets.map(t => ({ ...t, depends_on: [...new Set([...(t.depends_on || []), ...(t.blocked_by || [])])] }))
  let pending = tickets.slice()
  while (pending.length) {
    const gate = await mainCiGreen(s, phaseName)
    if (!gate.ok) {
      // a red main stops FEATURE tickets only; planned ci_fix tickets still run (CI FIRST; PR #270, QA-270-02)
      const fixes = pending.filter(t => t.ci_fix && (t.depends_on || []).every(d => tickets.some(x => x.id === d) ? (done.has(d) && done.get(d).merged) : true))
      if (!fixes.length) { pending.forEach(t => done.set(t.id, { id: t.id, merged: false, reason: `not started: ${gate.reason}` })); break }
      pending = pending.filter(t => !fixes.includes(t))
      log(`Sprint ${s.n}: main red; running ci_fix tickets only: ${fixes.map(t => t.id).join(', ')}`)
      const fr = await parallel(fixes.slice(0, WIP).map(t => () => deliverTicket(s, t, phaseName)))
      fr.forEach((r, i) => done.set(fixes[i].id, r || { id: fixes[i].id, merged: false, reason: 'agent failure' }))
      pending.push(...fixes.slice(WIP))
      continue
    }
    for (const t of pending) for (const d of t.depends_on) if (!tickets.some(x => x.id === d)) await mergedOnMain(d, phaseName)
    const ready = pending.filter(t => (t.depends_on || []).every(d => tickets.some(x => x.id === d) ? (done.has(d) && done.get(d).merged) : extMerged.get(d)) && (t.soft_deps || []).every(d => done.has(d) || !tickets.some(x => x.id === d)))
    const blocked = pending.filter(t => (t.depends_on || []).some(d => (done.has(d) && !done.get(d).merged) || (!tickets.some(x => x.id === d) && extMerged.get(d) === false)))
    blocked.forEach(t => done.set(t.id, { id: t.id, merged: false, reason: `blocked: Blocked-by ticket not merged into main (${t.depends_on.join(', ')})` }))
    pending = pending.filter(t => !blocked.includes(t))
    if (!ready.length) { pending.forEach(t => done.set(t.id, { id: t.id, merged: false, reason: 'unresolvable dependency' })); break }
    const wave = ready.slice(0, WIP) // WIP limit (ADR 0056): at most WIP tickets in flight
    pending = pending.filter(t => !wave.includes(t))
    log(`Sprint ${s.n}: wave of ${wave.length} (WIP ${WIP}): ${wave.map(t => t.id).join(', ')}`)
    const res = await parallel(wave.map(t => () => deliverTicket(s, t, phaseName)))
    res.forEach((r, i) => done.set(wave[i].id, r || { id: wave[i].id, merged: false, reason: 'agent failure' }))
  }
  return [...done.values()]
}
